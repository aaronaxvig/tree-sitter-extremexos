LLDP timers and management address
================================================================================

configure lldp transmit-interval 30
configure lldp transmit-delay auto
configure lldp management-address vlan Example secondary-ip 192.0.2.1

---

(source_file
  (command
    (configure_command
      (configure_lldp
        (configure_lldp_transmit_interval seconds: (word_or_string (word))))))
  (command
    (configure_command
      (configure_lldp
        (configure_lldp_transmit_delay (configure_lldp_transmit_delay_auto)))))
  (command
    (configure_command
      (configure_lldp
        (configure_lldp_management_address
          (configure_lldp_management_address_vlan
            vlan: (word_or_string (word))
            (configure_lldp_management_address_secondary_ip
              secondary_ip_address: (word_or_string (word)))))))))
